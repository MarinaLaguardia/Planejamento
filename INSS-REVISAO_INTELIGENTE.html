async function addSubject(){ const matter=document.getElementById("matter").value.trim(); const subject=document.getElementById("subject").value.trim(); const date=document.getElementById("date").value; const available=+document.getElementById("available").value; const done=+document.getElementById("done").value; const correct=+document.getElementById("correct").value; const difficulty=+document.getElementById("difficulty").value; const priority=+document.getElementById("priority").value; if(!matter||!subject||!date){ alert("Preencha matéria, assunto e data."); return; } if(done<0||correct<0||available<0||correct>done){ alert("Confira as quantidades de questões e acertos."); return; } const score=calcScore(done,correct); const days=interval(score,done,difficulty,priority); const next=addDays(date,days); const payload={ materia:matter, assunto:subject, data_conclusao:date, questoes_disponiveis:available, questoes_feitas:done, acertos:correct, dificuldade:difficulty, prioridade:priority, proxima_revisao:next, indice_dominio:score, ativo:true, questoes_necessarias:Math.max(0,10-done), nivel_confianca:score>=85?"alta":score>=70?"media":"baixa" }; const {error}=await db.from("assuntos").insert(payload); if(error){ console.error("Erro ao salvar:",error); showMessage("Não foi possível salvar: "+error.message,"error"); return; } document.getElementById("matter").value=""; document.getElementById("subject").value=""; document.getElementById("available").value="0"; document.getElementById("done").value="10"; document.getElementById("correct").value="0"; showMessage("Assunto salvo no Supabase. Próxima revisão: "+next); render(); }
